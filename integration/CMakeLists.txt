cmake_minimum_required(VERSION 3.22)
project(UrovInjectionTest C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
add_subdirectory(../Dobby dobby EXCLUDE_FROM_ALL)
add_library(uzhost SHARED host.c target.S)
target_compile_options(uzhost PRIVATE -O2 -Wall -Wextra -Werror -fno-omit-frame-pointer)
target_link_libraries(uzhost PRIVATE dl log z)
add_library(uzpayload SHARED payload.c ../api/src/urov.c)
target_include_directories(uzpayload PRIVATE ../api/include)
target_compile_options(uzpayload PRIVATE -O2 -Wall -Wextra -Werror)
target_link_libraries(uzpayload PRIVATE dobby_static log dl)
add_library(uzprobe SHARED probe.c)
target_compile_options(uzprobe PRIVATE -O2 -Wall -Wextra -Werror)
foreach(target uzhost uzpayload uzprobe)
    target_link_options(${target} PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,now)
endforeach()
add_executable(relro-test relro_test.c ../module/kclf/kclf_elf.c)
target_include_directories(relro-test PRIVATE ../module/kclf)
target_compile_definitions(relro-test PRIVATE KCLF_ELF_HOST)
target_compile_options(relro-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
